1. Import auth helpers (requireAuth/requireEditor) and company helper
2. Import the shared handler from `@/lib/financeiro-shared`
3. Export GET/PUT/PATCH that do auth check, get company, then delegate to shared handler

Example for contas-pagar/invoice/[invoiceId]/installments/route.ts:
```typescript
import { NextResponse } from 'next/server';
import { requireEditor, unauthorizedResponse, forbiddenResponse } from '@/lib/auth';
import { getOrCreateSingleCompany } from '@/lib/single-company';
import { handleInstallmentsGet, handleInstallmentsPut } from '@/lib/financeiro-shared';

export async function GET(req: Request, { params }: { params: { invoiceId: string } }) {
  const session = await requireEditor();
  if (!session) return unauthorizedResponse();
  const company = await getOrCreateSingleCompany();
  if (!company) return forbiddenResponse();
  return handleInstallmentsGet(params.invoiceId, company);
}

export async function PUT(req: Request, { params }: { params: { invoiceId: string } }) {
  const session = await requireEditor();
  if (!session) return unauthorizedResponse();
  const company = await getOrCreateSingleCompany();
  if (!company) return forbiddenResponse();
  const body = await req.json();
  return handleInstallmentsPut(params.invoiceId, company, body);
}
```

For contas-receber versions, same pattern but pass `'receber'` as direction where needed.
