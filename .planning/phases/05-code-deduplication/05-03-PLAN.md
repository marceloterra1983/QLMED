   - CNPJ field: suppliers use `senderCnpj`/`senderName`, customers use `recipientCnpj`/`recipientName`

   - Product count: both use invoice_item_tax but with different direction filters

```typescript

    xmlPartyPath: 'emit',    // XML path for party data
  },
  customer: {
    invoiceDirection: 'issued' as const,
    cnpjField: 'recipientCnpj' as const,
    nameField: 'recipientName' as const,

  },
} as const;
```
